cmake_minimum_required(VERSION 3.20)
project(KHyperSystem LANGUAGES CXX)
add_library(khypergraph STATIC backend/native/vendor/khypergraph/src/hyper_graph.cpp)
target_include_directories(khypergraph PUBLIC backend/native/vendor/khypergraph/include)
target_compile_features(khypergraph PUBLIC cxx_std_20)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import pyarrow; print(pyarrow.__path__[0])"
                OUTPUT_VARIABLE ARROW_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE
                RESULT_VARIABLE ARROW_DISCOVERY)
if(NOT ARROW_DISCOVERY EQUAL 0)
  message(FATAL_ERROR "PyArrow is required; run setup.ps1 to install dependencies")
endif()
file(TO_CMAKE_PATH "${ARROW_ROOT}" ARROW_ROOT)
file(GLOB ARROW_RUNTIME_DLLS "${ARROW_ROOT}/../pyarrow.libs/*.dll")
if(NOT MSVC)
  message(FATAL_ERROR "The Windows PyArrow C++ libraries require MSVC; use setup.ps1")
endif()
add_library(arrow_shared SHARED IMPORTED)
set_target_properties(arrow_shared PROPERTIES IMPORTED_IMPLIB "${ARROW_ROOT}/arrow.lib"
                      IMPORTED_LOCATION "${ARROW_ROOT}/arrow.dll")
add_library(parquet_shared SHARED IMPORTED)
set_target_properties(parquet_shared PROPERTIES IMPORTED_IMPLIB "${ARROW_ROOT}/parquet.lib"
                      IMPORTED_LOCATION "${ARROW_ROOT}/parquet.dll")
add_executable(khyper_worker backend/native/src/worker.cpp backend/native/src/dataset_reader.cpp backend/native/src/query_engine.cpp backend/native/src/leiden.cpp)
target_link_libraries(khyper_worker PRIVATE khypergraph arrow_shared parquet_shared)
target_include_directories(khyper_worker SYSTEM PRIVATE "${ARROW_ROOT}/include")
add_custom_command(TARGET khyper_worker POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ARROW_ROOT}/arrow.dll" "$<TARGET_FILE_DIR:khyper_worker>"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ARROW_ROOT}/parquet.dll" "$<TARGET_FILE_DIR:khyper_worker>")
if(ARROW_RUNTIME_DLLS)
  add_custom_command(TARGET khyper_worker POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${ARROW_RUNTIME_DLLS} "$<TARGET_FILE_DIR:khyper_worker>")
endif()
target_include_directories(khyper_worker PRIVATE backend/native/vendor)
if(MINGW)
  target_link_options(khyper_worker PRIVATE -static -static-libgcc -static-libstdc++)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_core_validation tests/test_core_validation.cpp)
  target_link_libraries(test_core_validation PRIVATE khypergraph)
  if(MINGW)
    target_link_options(test_core_validation PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME core_validation COMMAND test_core_validation)
endif()
if(MSVC)
  target_compile_options(khyper_worker PRIVATE /utf-8 /W4)
else()
  target_compile_options(khyper_worker PRIVATE -Wall -Wextra -Wpedantic)
endif()
